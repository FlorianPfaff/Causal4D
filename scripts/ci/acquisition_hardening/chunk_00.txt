"""Apply the reviewed acquisition-evidence hardening patch idempotently."""

from __future__ import annotations

from pathlib import Path


ROOT = Path(__file__).resolve().parents[2]


def _replace(path: str, old: str, new: str) -> None:
    target = ROOT / path
    text = target.read_text(encoding="utf-8")
    count = text.count(old)
    if count == 0 and new in text:
        return
    if count != 1:
        if path == "src/causal4d/contracts.py" and old.startswith(
            "def save_contract("
        ):
            start = text.index("def save_contract(")
            stop = text.index("\ndef load_contract(", start)
            replacement = new.rstrip() + "\n\n"
            target.write_text(
                text[:start] + replacement + text[stop + 1 :],
                encoding="utf-8",
            )
            return
        raise RuntimeError(
            f"expected one replacement in {path}, found {count}; "
            f"anchor={old[:180]!r}"
        )
    target.write_text(text.replace(old, new), encoding="utf-8")


def _append(path: str, marker: str, addition: str) -> None:
    target = ROOT / path
    text = target.read_text(encoding="utf-8")
    if marker in text:
        return
    target.write_text(text.rstrip() + "\n\n" + addition.strip() + "\n", encoding="utf-8")


def _atomic_io() -> None:
    _replace(
        "src/causal4d/atomic_io.py",
        "import tempfile\nfrom pathlib import Path\nfrom typing import Any\n",
        "import tempfile\nfrom collections.abc import Callable\nfrom pathlib import Path\nfrom typing import Any, BinaryIO\n",
    )
    _replace(
        "src/causal4d/atomic_io.py",
        "\ndef atomic_write_text(\n",
        '''\ndef atomic_write_binary(\n    path: str | Path,\n    writer: Callable[[BinaryIO], None],\n    *,\n    overwrite: bool = True,\n    validate: Callable[[Path], None] | None = None,\n) -> None:\n    """Publish a binary artifact atomically after an optional validation pass."""\n\n    target = Path(path)\n    target.parent.mkdir(parents=True, exist_ok=True)\n    descriptor, temporary_name = tempfile.mkstemp(\n        prefix=f".{target.name}.",\n        suffix=".tmp",\n        dir=target.parent,\n    )\n    temporary = Path(temporary_name)\n    try:\n        with os.fdopen(descriptor, "w+b") as handle:\n            writer(handle)\n            handle.flush()\n            os.fsync(handle.fileno())\n        if validate is not None:\n            validate(temporary)\n        if overwrite:\n            os.replace(temporary, target)\n        else:\n            os.link(temporary, target)\n            temporary.unlink()\n        _fsync_directory(target.parent)\n    finally:\n        temporary.unlink(missing_ok=True)\n\n\ndef atomic_write_text(\n''',
    )
    _replace(
        "src/causal4d/atomic_io.py",
        '__all__ = ["atomic_write_json", "atomic_write_text"]\n',
        '''__all__ = [\n    "atomic_write_binary",\n    "atomic_write_json",\n    "atomic_write_text",\n]\n''',
    )
